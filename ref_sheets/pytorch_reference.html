Model Definition

import torch
import torch.nn as nn

# Simple MLP
# NOTE: sub-modules must be stored as nn.Module attributes, nn.Sequential,
# nn.ModuleList, or nn.ModuleDict — NOT plain Python lists/dicts.
# Plain list/dict won't register parameters, so optimizer won't see them.
class MLP(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, output_dim),
        )
        # nn.ModuleList — use when you need to iterate over layers dynamically
        # self.layers = nn.ModuleList([nn.Linear(hidden_dim, hidden_dim) for _ in range(n)])
        # nn.ModuleDict — use when you want named access
        # self.heads = nn.ModuleDict({'cls': nn.Linear(hidden_dim, 10), 'reg': nn.Linear(hidden_dim, 1)})

    def forward(self, x):
        return self.net(x)

model = MLP(784, 128, 10)

MLP for mixed continuous + categorical data

class HybridMLP(nn.Module):
    def __init__(self, n_cont, cat_sizes, emb_dim=8, hidden=64, n_classes=2):
        super().__init__()
        # One embedding per categorical feature (learns dense repr for each category)
        self.embeddings = nn.ModuleList([
            nn.Embedding(n_cats, emb_dim) for n_cats in cat_sizes
        ])
        input_dim = n_cont + len(cat_sizes) * emb_dim
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden), nn.ReLU(),
            nn.Linear(hidden, hidden), nn.ReLU(),
            nn.Linear(hidden, n_classes),
        )

    def forward(self, x_cont, x_cat):
        # x_cont: (batch, n_cont) float   x_cat: (batch, n_cat_features) long
        embs = [emb(x_cat[:, i]) for i, emb in enumerate(self.embeddings)]
        x = torch.cat([x_cont] + embs, dim=-1)
        return self.net(x)

What Optimizers Actually Do

# After loss.backward(), each parameter has a .grad attribute.
# Manual SGD (modern idiom — wrap in no_grad to skip autograd tracking):
with torch.no_grad():
    for param in model.parameters():
        if param.grad is not None:        # grads are None before first .backward()
            param.sub_(lr * param.grad)   # in-place subtract
# Then reset grads (set to None is faster than zeroing — frees memory):
for param in model.parameters():
    param.grad = None
# optimizer.step() does the update; optimizer.zero_grad() resets grads
# (defaults to set_to_none=True since PyTorch 2.0).

Training Loop

import torch.optim as optim

model = MLP(784, 128, 10)
optimizer = optim.AdamW(model.parameters(), lr=1e-3, weight_decay=1e-4)
loss_fn = nn.CrossEntropyLoss()

for epoch in range(num_epochs):
    model.train()
    for batch_x, batch_y in train_loader:
        optimizer.zero_grad()
        logits = model(batch_x)
        loss = loss_fn(logits, batch_y)
        loss.backward()
        optimizer.step()

    # Validation
    model.eval()
    with torch.no_grad():
        val_loss = 0
        correct = 0
        total = 0
        for batch_x, batch_y in val_loader:
            logits = model(batch_x)
            val_loss += loss_fn(logits, batch_y).item() * batch_x.size(0)
            preds = logits.argmax(dim=-1)
            correct += (preds == batch_y).sum().item()
            total += batch_y.size(0)
        val_loss /= total
        val_acc = correct / total

Reusable Training Function (with tqdm + logging)

from tqdm.auto import tqdm  # auto: notebook bar in Jupyter, terminal bar elsewhere

def train(model, train_loader, val_loader, epochs, lr=1e-3,
          device='cpu', verbose=True):
    model = model.to(device)
    opt = optim.Adam(model.parameters(), lr=lr)
    loss_fn = nn.CrossEntropyLoss()
    history = []  # list of per-epoch dicts → pd.DataFrame(history) for analysis

    pbar = tqdm(range(epochs), disable=not verbose)
    for epoch in pbar:
        # --- Train ---
        model.train()
        train_loss_sum, n_train = 0.0, 0
        for x, y in train_loader:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            loss = loss_fn(model(x), y)
            loss.backward()
            opt.step()
            train_loss_sum += loss.item() * x.size(0)  # weight by batch size
            n_train += x.size(0)
        train_loss = train_loss_sum / n_train

        # --- Validate ---
        model.eval()
        val_loss_sum, n_correct, n_val = 0.0, 0, 0
        with torch.no_grad():
            for x, y in val_loader:
                x, y = x.to(device), y.to(device)
                logits = model(x)
                val_loss_sum += loss_fn(logits, y).item() * x.size(0)
                n_correct += (logits.argmax(-1) == y).sum().item()
                n_val += x.size(0)
        val_loss = val_loss_sum / n_val
        val_acc = n_correct / n_val

        history.append({
            "epoch": epoch,
            "train_loss": train_loss,
            "val_loss": val_loss,
            "val_acc": val_acc,
        })
        pbar.set_postfix(train=f"{train_loss:.4f}",
                         val=f"{val_loss:.4f}",
                         acc=f"{val_acc:.4f}")
    return history

Early stopping (drop-in for the train loop)

best_val_loss = float('inf')
wait = 0
patience = 5

for epoch in range(num_epochs):
    # ... train ...
    # ... compute val_loss ...

    if val_loss < best_val_loss:
        best_val_loss = val_loss
        wait = 0
        torch.save(model.state_dict(), 'best_model.pt')  # checkpoint best
    else:
        wait += 1
        if wait >= patience:
            break  # no improvement for `patience` epochs → stop

model.load_state_dict(torch.load('best_model.pt'))  # restore best weights

# Adam's `weight_decay` arg is L2 penalty (added to the gradient, then routed
# through Adam's adaptive scaling) — not true decoupled decay. AdamW is the
# corrected version: it applies the decay directly to the weights, outside
# the adaptive update. Prefer AdamW whenever you want weight decay with Adam.

Gotchas

    output = model(x)

# - eval() disables dropout, changes batchnorm behavior
# - train() re-enables them

#   input: raw logits (batch, num_classes) — NOT softmax
#   target: either class indices (batch,) as LongTensor,
#           OR class probabilities (batch, num_classes) as float
#           (one-hot / soft labels supported since PyTorch 1.10)

# clone() vs detach()
# - detach(): removes from computation graph (shares data)
# - clone(): copies data (stays in graph)
# - detach().clone(): independent copy, no grad tracking

Simple Transformer (from scratch)

class SelfAttention(nn.Module):
    def __init__(self, d_model, n_heads):
        super().__init__()
        self.n_heads = n_heads
        self.d_head = d_model // n_heads
        self.qkv = nn.Linear(d_model, 3 * d_model)
        self.out = nn.Linear(d_model, d_model)

    def forward(self, x):
        B, T, C = x.shape
        qkv = self.qkv(x).reshape(B, T, 3, self.n_heads, self.d_head)
        qkv = qkv.permute(2, 0, 3, 1, 4)  # (3, B, heads, T, d_head)
        q, k, v = qkv[0], qkv[1], qkv[2]

        attn = (q @ k.transpose(-2, -1)) / (self.d_head ** 0.5)
        # Optional causal mask:
        # mask = torch.triu(torch.ones(T, T, device=x.device), diagonal=1).bool()
        # attn = attn.masked_fill(mask, float('-inf'))
        attn = attn.softmax(dim=-1)

        out = (attn @ v).transpose(1, 2).reshape(B, T, C)
        return self.out(out), attn  # return attn weights for analysis


class TransformerBlock(nn.Module):
    def __init__(self, d_model, n_heads):
        super().__init__()
        self.attn = SelfAttention(d_model, n_heads)
        self.ln1 = nn.LayerNorm(d_model)
        self.ln2 = nn.LayerNorm(d_model)
        self.mlp = nn.Sequential(
            nn.Linear(d_model, 4 * d_model),
            nn.GELU(),
            nn.Linear(4 * d_model, d_model),
        )

    def forward(self, x):
        attn_out, attn_weights = self.attn(self.ln1(x))
        x = x + attn_out
        x = x + self.mlp(self.ln2(x))
        return x, attn_weights


class SmallTransformer(nn.Module):
    def __init__(self, vocab_size, d_model, n_heads, n_layers, max_seq_len):
        super().__init__()
        self.tok_emb = nn.Embedding(vocab_size, d_model)
        self.pos_emb = nn.Embedding(max_seq_len, d_model)
        self.blocks = nn.ModuleList(
            [TransformerBlock(d_model, n_heads) for _ in range(n_layers)]
        )
        self.ln_f = nn.LayerNorm(d_model)
        self.head = nn.Linear(d_model, vocab_size)

    def forward(self, x):
        B, T = x.shape
        pos = torch.arange(T, device=x.device)
        x = self.tok_emb(x) + self.pos_emb(pos)

        all_attn = []
        for block in self.blocks:
            x, attn = block(x)
            all_attn.append(attn)

        x = self.ln_f(x)
        logits = self.head(x)
        return logits, all_attn

Transformer (using built-in nn components)

class BuiltinTransformer(nn.Module):
    def __init__(self, vocab_size, d_model, n_heads, n_layers, max_seq_len, dim_ff=None):
        super().__init__()
        dim_ff = dim_ff or 4 * d_model
        self.tok_emb = nn.Embedding(vocab_size, d_model)
        self.pos_emb = nn.Embedding(max_seq_len, d_model)
        # nn.TransformerEncoderLayer = self-attn + feedforward + layernorm + residuals
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=n_heads, dim_feedforward=dim_ff,
            batch_first=True,  # input/output shape: (batch, seq, d_model)
            norm_first=True,   # pre-norm (like GPT) vs post-norm
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)
        self.ln_f = nn.LayerNorm(d_model)
        self.head = nn.Linear(d_model, vocab_size)

    def forward(self, x, causal=True):
        B, T = x.shape
        pos = torch.arange(T, device=x.device)
        x = self.tok_emb(x) + self.pos_emb(pos)
        # is_causal=True alone enables PyTorch 2.0+ SDPA fast path (FlashAttention etc.).
        # Passing an explicit mask alongside can disable that fast path — only do that
        # if you need a non-standard mask shape.
        x = self.transformer(x, is_causal=causal)
        return self.head(self.ln_f(x))
