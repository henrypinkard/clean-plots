Selecting / Filtering

# Column selection
df['col']                  # Series
df[['col1', 'col2']]      # DataFrame

# Row filtering
df[df['accuracy'] > 0.9]
df[df['model'].isin(['RF', 'GB'])]
df[(df['lr'] == 0.01) & (df['accuracy'] > 0.9)]  # must use & not 'and'
df.query('lr == 0.01 and accuracy > 0.9')          # string query (cleaner)

# loc — select by row LABELS and column NAMES (use ~90% of the time)
# iloc — select by integer POSITIONS, like numpy (for "first 10 rows", "last row", etc.)
# Both return views or copies depending on context; safe for reading, use for assignment too.
df.loc[0:5, 'accuracy']                    # rows with labels 0–5 (inclusive!), column 'accuracy'
df.iloc[0:5, 1]                             # rows at positions 0–4 (exclusive end), 2nd column
df.loc[df['model'] == 'RF', 'accuracy']     # all RF rows, just the 'accuracy' column
df.loc[df['model'] == 'RF', 'accuracy'] = 0.99  # can also assign through loc

Sorting

# Returns a NEW DataFrame (does NOT modify in place). Index stays with the rows.
df.sort_values('accuracy', ascending=False)
df.sort_values(['model', 'accuracy'], ascending=[True, False])
# To reset the index to 0,1,2,... after sorting:
df.sort_values('accuracy', ascending=False).reset_index(drop=True)

Adding / Modifying Columns

df['new_col'] = df['col1'] + df['col2']
df['log_loss'] = np.log(df['loss'])
df['category'] = df['value'].apply(lambda x: 'high' if x > 0.5 else 'low')
df['rank'] = df['accuracy'].rank(ascending=False)  # 1.0 = highest accuracy, ties get avg rank

# top 3 runs per model (rank + groupby, no apply needed)
df[df.groupby('model')['accuracy'].rank(ascending=False) <= 3]

GroupBy + Aggregation

# Single aggregation
df.groupby('model')['accuracy'].mean()

# Multiple aggs on one column — gives MultiIndex columns: ('accuracy','mean'), ('accuracy','std')
df.groupby('model')['accuracy'].agg(['mean', 'std', 'count'])

# PREFER THIS: named output columns, flat index, no ambiguity
df.groupby('model').agg(
    mean_acc=('accuracy', 'mean'),
    std_acc=('accuracy', 'std'),
    mean_time=('train_time', 'mean'),
)

# GroupBy with multiple keys
df.groupby(['model', 'lr'])['accuracy'].mean()

# Reset index after groupby (gets back a flat DataFrame)
df.groupby('model')['accuracy'].mean().reset_index()

agg vs transform vs apply

# agg: many rows → one row per group (shrinks)
df.groupby('model')['accuracy'].agg(['mean', 'std'])

# transform: same shape back — broadcasts group-level result to every row
df['acc_zscore'] = df.groupby('model')['accuracy'].transform(
    lambda x: (x - x.mean()) / x.std()
)

# apply: escape hatch — lambda receives a DataFrame (the sub-table for each group)
df.groupby('model').apply(lambda group_df: group_df.nlargest(3, 'accuracy'))

Collecting results across runs

results = []
for model_name, model in models.items():

        model.fit(X_train, y_train)
        acc = accuracy_score(y_test, model.predict(X_test))

Experiment grid with pivot

results = []
for n_train in [100, 500, 1000, 5000]:
    for model_name in ['RF', 'GB', 'LR']:
        for seed in range(3):
            acc = run_experiment(model_name, n_train, seed)
            results.append({'model': model_name, 'n_train': n_train,
                           'seed': seed, 'accuracy': acc})
df = pd.DataFrame(results)

# Pivot: rows=n_train, cols=model, values=mean accuracy
pivot = df.pivot_table(values='accuracy', index='n_train', columns='model', aggfunc='mean')
pivot.plot(marker='o')  # quick line plot

Useful Tricks

# String formatting in output
pd.set_option('display.float_format', '{:.4f}'.format)

# Temporarily show all rows (or more cols) without changing global settings
with pd.option_context('display.max_rows', 1000):
    display(df)
# Other options: 'display.max_columns', 'display.width', 'display.max_colwidth'

# Apply to all numeric columns
df.select_dtypes(include='number')

# Rename columns
df.rename(columns={'old_name': 'new_name'})

# Drop columns
df.drop(columns=['unwanted'])

# Melt (wide → tidy: one row per observation)
# wide:   model  epoch_0  epoch_1  epoch_2
#         mlp    0.9      0.7      0.5
pd.melt(df, id_vars='model', var_name='epoch', value_name='loss')
# tidy:   model  epoch    loss
#         mlp    epoch_0  0.9
#         mlp    epoch_1  0.7
#         mlp    epoch_2  0.5

# json_normalize (flatten nested dicts into columns)
data = {"model": "mlp", "metrics": {"train": {"loss": 0.5}, "val": {"loss": 0.6}}}
pd.json_normalize(data)
#   model  metrics.train.loss  metrics.val.loss
#   mlp    0.5                 0.6
# Both are "get your data tidy" steps before the pivot/aggregate workflow kicks in.

# DataFrame → markdown table (paste into slides, GitHub, notebooks)
print(df.to_markdown(index=False))               # requires `pip install tabulate`
print(df.to_markdown(index=False, floatfmt='.3f'))  # control number formatting
